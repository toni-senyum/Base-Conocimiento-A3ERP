// Indicar detalle artículo con DUS
public class LineaDetalleParametros : ILineaDetalleParametros
{
    public DateTime FechaCaducidad { get; set; }
    public int Linea { get; set; }
    public string Lote { get; set; }
    public string NumeroSerie { get; set; }
    public string Ubicacion { get; set; }
    public double UnidadesStock { get; set; }
    public double UnidadesVenta { get; set; }
}

// Uno con lote y otro con fechas
LineaDetalleParametros param = new LineaDetalleParametros();

NaxItem.NuevaLinea();
NaxItem.AsStringLin["CODART"] = "L2";
NaxItem.AsFloatLin["UNIDADESSTOCK"] = 3;
NaxItem.AsFloatLin["UNIDADES"] = 3;
param.Lote = "l1";
param.UnidadesStock = 3;
param.UnidadesVenta = 3;
((ILineaDetalle)NaxItem).AnadirDetalle(param);
NaxItem.AnadirLinea();
